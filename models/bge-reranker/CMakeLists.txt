include(${CMAKE_SOURCE_DIR}/tools/buddy-codegen/cmake/buddy_model.cmake)
include(ProcessorCount)
ProcessorCount(_BUDDY_BGE_RERANKER_DEFAULT_COMPILE_JOBS)
if(_BUDDY_BGE_RERANKER_DEFAULT_COMPILE_JOBS EQUAL 0)
  set(_BUDDY_BGE_RERANKER_DEFAULT_COMPILE_JOBS 1)
endif()
set(BUDDY_BGE_RERANKER_SPEC "${CMAKE_CURRENT_SOURCE_DIR}/specs/f32.json" CACHE FILEPATH "Variant spec JSON")
set(BUDDY_BGE_RERANKER_HF_CONFIG "" CACHE FILEPATH "Optional HF config.json")
set(BUDDY_BGE_RERANKER_LOCAL_MODEL "" CACHE PATH "Optional local HF model dir")
set(BUDDY_BGE_RERANKER_COMPILE_JOBS "${_BUDDY_BGE_RERANKER_DEFAULT_COMPILE_JOBS}" CACHE STRING "Parallel jobs")
set(BGE_RERANKER_BUILD_DIR "" CACHE PATH "Mode A: pre-built .o")
set(BGE_RERANKER_MLIR_DIR "" CACHE PATH "Mode B: pre-generated MLIR")
if(NOT Python3_EXECUTABLE)
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
endif()
buddy_add_model(NAME bge_reranker SPEC "${BUDDY_BGE_RERANKER_SPEC}" RUNNER_SRC Bge_rerankerRunner.cpp
  HF_CONFIG "${BUDDY_BGE_RERANKER_HF_CONFIG}" LOCAL_MODEL "${BUDDY_BGE_RERANKER_LOCAL_MODEL}"
  BUILD_DIR "${BGE_RERANKER_BUILD_DIR}" MLIR_DIR "${BGE_RERANKER_MLIR_DIR}"
  COMPILE_JOBS "${BUDDY_BGE_RERANKER_COMPILE_JOBS}")
