include(${CMAKE_SOURCE_DIR}/tools/buddy-codegen/cmake/buddy_model.cmake)
include(ProcessorCount)
ProcessorCount(_BUDDY_SAM2_DEFAULT_COMPILE_JOBS)
if(_BUDDY_SAM2_DEFAULT_COMPILE_JOBS EQUAL 0) set(_BUDDY_SAM2_DEFAULT_COMPILE_JOBS 1) endif()
set(BUDDY_SAM2_SPEC "${CMAKE_CURRENT_SOURCE_DIR}/specs/f32.json" CACHE FILEPATH "Variant spec")
set(BUDDY_SAM2_HF_CONFIG "" CACHE FILEPATH "Optional HF config.json")
set(BUDDY_SAM2_LOCAL_MODEL "" CACHE PATH "Optional local HF model dir")
set(BUDDY_SAM2_COMPILE_JOBS "${_BUDDY_SAM2_DEFAULT_COMPILE_JOBS}" CACHE STRING "Parallel jobs")
set(SAM2_BUILD_DIR "" CACHE PATH "Mode A"); set(SAM2_MLIR_DIR "" CACHE PATH "Mode B")
if(NOT Python3_EXECUTABLE) find_package(Python3 COMPONENTS Interpreter REQUIRED) endif()
buddy_add_model(NAME sam2 SPEC "${BUDDY_SAM2_SPEC}" RUNNER_SRC SAM2Runner.cpp
  HF_CONFIG "${BUDDY_SAM2_HF_CONFIG}" LOCAL_MODEL "${BUDDY_SAM2_LOCAL_MODEL}"
  BUILD_DIR "${SAM2_BUILD_DIR}" MLIR_DIR "${SAM2_MLIR_DIR}" COMPILE_JOBS "${BUDDY_SAM2_COMPILE_JOBS}")
