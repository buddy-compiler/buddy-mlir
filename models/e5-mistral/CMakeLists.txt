include(${CMAKE_SOURCE_DIR}/tools/buddy-codegen/cmake/buddy_model.cmake)
include(ProcessorCount)
ProcessorCount(_BUDDY_E5_MISTRAL_DEFAULT_COMPILE_JOBS)
if(_BUDDY_E5_MISTRAL_DEFAULT_COMPILE_JOBS EQUAL 0) set(_BUDDY_E5_MISTRAL_DEFAULT_COMPILE_JOBS 1) endif()
set(BUDDY_E5_MISTRAL_SPEC "${CMAKE_CURRENT_SOURCE_DIR}/specs/f32.json" CACHE FILEPATH "Variant spec")
set(BUDDY_E5_MISTRAL_HF_CONFIG "" CACHE FILEPATH "Optional HF config.json")
set(BUDDY_E5_MISTRAL_LOCAL_MODEL "" CACHE PATH "Optional local HF model dir")
set(BUDDY_E5_MISTRAL_COMPILE_JOBS "${_BUDDY_E5_MISTRAL_DEFAULT_COMPILE_JOBS}" CACHE STRING "Parallel jobs")
set(E5_MISTRAL_BUILD_DIR "" CACHE PATH "Mode A: pre-built .o")
set(E5_MISTRAL_MLIR_DIR "" CACHE PATH "Mode B: pre-generated MLIR")
if(NOT Python3_EXECUTABLE) find_package(Python3 COMPONENTS Interpreter REQUIRED) endif()
buddy_add_model(NAME e5_mistral SPEC "${BUDDY_E5_MISTRAL_SPEC}" RUNNER_SRC E5MistralRunner.cpp
  HF_CONFIG "${BUDDY_E5_MISTRAL_HF_CONFIG}" LOCAL_MODEL "${BUDDY_E5_MISTRAL_LOCAL_MODEL}"
  BUILD_DIR "${E5_MISTRAL_BUILD_DIR}" MLIR_DIR "${E5_MISTRAL_MLIR_DIR}" COMPILE_JOBS "${BUDDY_E5_MISTRAL_COMPILE_JOBS}")
